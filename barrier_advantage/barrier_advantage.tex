\definecolor{countcolor}{RGB}{50,50,255}
\definecolor{seqcolor}{RGB}{150,0,0}
\definecolor{fullcolor}{RGB}{60,120,110}
\def\DashGap{1.50pt}
% Half-length of the horizontal cap in PGF coordinates (before final resizing).
\def\ErrorCapSize{1.15pt}
\pgfdeclarelayer{reference}
\pgfdeclarelayer{data}
\pgfplotsset{
  layers/panel/.define layer set={
    axis background,axis grid,pre main,reference,main,data,
    axis ticks,axis lines,axis tick labels,axis descriptions,axis foreground
  }{/pgfplots/layers/standard},
  curve/.style={line width=0.9pt,on layer=data},
  % The error-mark node is sloped along the vertical error bar. Rotate the
  % local '-' by 90 degrees to obtain HORIZONTAL caps at both endpoints.
  % Keep the shaft and caps solid even on dashed/dash-dot data curves.
  mc error/.style={error bars/y dir=both,error bars/y explicit,
    error bars/error mark=-,
    error bars/error mark options={rotate=90,mark size=\ErrorCapSize,
      solid,line width=0.35pt,line cap=butt},
    error bars/error bar style={solid,line width=0.35pt,line cap=butt}},
  N curve/.style={curve,countcolor,mark=*,mark size=0.85pt,
    dash pattern=on 4.5pt off \DashGap on 0.01pt off \DashGap,
    line cap=round,
    mark options={fill=white,draw=countcolor,line width=0.4pt}},
  S curve/.style={curve,seqcolor,mark=square*,mark size=0.88pt,
    dash pattern=on 2.4pt off \DashGap,
    mark options={fill=white,draw=seqcolor,line width=0.4pt}},
  X curve/.style={curve,fullcolor,mark=diamond*,mark size=1.12pt,solid,
    mark options={fill=white,draw=fullcolor,line width=0.4pt}},
}
